cmake_minimum_required(VERSION 3.28)
project(obs-webview2-drop-tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets)
set(WEB_SOURCE ../../../frontend/webview2)
add_executable(drop-test ../drop-test.cpp ${WEB_SOURCE}/ExternalDrop.cpp ${WEB_SOURCE}/BridgeProtocol.cpp)
target_link_libraries(drop-test PRIVATE Qt6::Widgets)
target_include_directories(drop-test PRIVATE ${WEB_SOURCE})
enable_testing()
add_test(NAME drop-validation COMMAND drop-test)
include(../../../cmake/windows/webview2.cmake)
obs_find_webview2()
add_executable(drop-host-test ../drop-host-test.cpp ${WEB_SOURCE}/WebView2Widget.cpp ${WEB_SOURCE}/WebView2Widget.hpp
  ${WEB_SOURCE}/ExternalDrop.cpp ${WEB_SOURCE}/BridgeProtocol.cpp)
target_link_libraries(drop-host-test PRIVATE Qt6::Widgets OBS::WebView2)
target_include_directories(drop-host-test PRIVATE ${WEB_SOURCE})
target_compile_definitions(drop-host-test PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
add_test(NAME drop-host COMMAND drop-host-test "${CMAKE_CURRENT_SOURCE_DIR}/../drop-fixture")
