cmake_minimum_required(VERSION 3.28)
project(obs-webview2-preview-mask-tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets)
set(WEB_SOURCE ../../../frontend/webview2)
include(../../../cmake/windows/webview2.cmake)
obs_find_webview2()
add_executable(preview-mask-host ../preview-mask-host.cpp ${WEB_SOURCE}/WebView2Widget.cpp
  ${WEB_SOURCE}/WebView2Widget.hpp ${WEB_SOURCE}/ExternalDrop.cpp ${WEB_SOURCE}/BridgeProtocol.cpp)
target_include_directories(preview-mask-host PRIVATE ${WEB_SOURCE})
target_link_libraries(preview-mask-host PRIVATE Qt6::Widgets OBS::WebView2)
target_compile_definitions(preview-mask-host PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
enable_testing()
add_test(NAME preview-mask-host COMMAND preview-mask-host "${CMAKE_CURRENT_SOURCE_DIR}/../drop-fixture")
